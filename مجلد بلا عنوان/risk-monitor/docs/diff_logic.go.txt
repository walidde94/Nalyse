package diff

import (
	"context"
	"fmt"
	"risk-monitor/internal/core"
)

// DiffEngine encapsulates the logic for comparing scans
type DiffEngine struct {
	repo Repository
}

// CompareAndPersist loads the previous completed scan and generates issues based on the diff.
// This is transactionally safe.
func (d *DiffEngine) CompareAndPersist(ctx context.Context, currentScan *core.Scan) error {
	// 1. Fetch Last Completed Scan
	lastScan, err := d.repo.GetLastCompletedScan(ctx, currentScan.WebsiteID)
	if err != nil {
		return err // Handle 'not found' as a first-run (no diff, just baseline)
	}

	// 2. Load Data Maps
	currentUrls := d.mapURLs(currentScan.URLs)
	lastUrls := d.mapURLs(lastScan.URLs)
	
	// 3. Diff URLs
	for url, curr := range currentUrls {
		prev, exists := lastUrls[url]
		
		// Regression: Was OK, Now Error
		if exists && prev.IsHealthy() && !curr.IsHealthy() {
			d.upsertIssue(ctx, core.Issue{
				Type: core.IssueTypeUrlError,
				Severity: core.SeverityCritical,
				Identifier: url,
				WebsiteID: currentScan.WebsiteID,
			})
		}
		
		// Recovery: Was Error, Now OK
		if exists && !prev.IsHealthy() && curr.IsHealthy() {
			d.resolveIssue(ctx, currentScan.WebsiteID, url, core.IssueTypeUrlError)
		}
	}

	// 4. Check for Asset Regressions
	// Similar logic for Assets...

	// 5. Check for Cookie Banner Regression
	// If banner was present and now missing -> Critical Issue
	if lastScan.CookieCheck.BannerDetected && !currentScan.CookieCheck.BannerDetected {
		d.upsertIssue(ctx, core.Issue{
			Type: core.IssueTypeCookieMissing,
			Severity: core.SeverityCritical,
			Identifier: "cookie_banner",
			WebsiteID: currentScan.WebsiteID,
		})
	}
	
	// 6. Update Scan Status
	// Calculate based on open issues count
	
	return nil
}
